<body>
  <h1>Piedra, Papel o Tijera</h1>
  <button onclick="jugar('piedra')">Piedra</button>
  <button onclick="jugar('papel')">Papel</button>
  <button onclick="jugar('tijera')">Tijera</button>
  <p id="resultado"></p>

  <script>
    function jugar(eleccionJugador) {
      const opciones = ['piedra', 'papel', 'tijera'];
      const eleccionPC = opciones[Math.floor(Math.random() * 3)];

      let resultado = '';
      if (eleccionJugador === eleccionPC) {
        resultado = 'Empate. Ambos eligieron ' + eleccionJugador;
      } else if (
        (eleccionJugador === 'piedra' && eleccionPC === 'tijera') ||
        (eleccionJugador === 'papel' && eleccionPC === 'piedra') ||
        (eleccionJugador === 'tijera' && eleccionPC === 'papel')
      ) {
        resultado = '¡Ganaste! La PC eligió ' + eleccionPC;
      } else {
        resultado = 'Perdiste. La PC eligió ' + eleccionPC;
      }

      document.getElementById('resultado').innerText = resultado;
    }
  </script>
</body>
